create database sql_join_exp_7;
use sql_join_exp_7;

create table Dept(
dept_no int primary key,
dept_name varchar(20) not null,
loc varchar(20)
);

desc Dept;

create table employee(
emp_id int primary key,
e_name varchar(20) not null,
job varchar(20),
mgr int,
hiredate date,
sal decimal(10,2),
comm decimal(10,2),
dept_no int,

constraint fk_emp_dept
foreign key(dept_no)
references Dept(dept_no)
);

desc employee;

create table salgrade(
grade int primary key,
losal decimal(10,2) not null,
hisal decimal(10,2) not null
);

desc salgrade;

create table areas(
area_id int primary key,
area_name varchar(20) not null,
location varchar(20)
);

select e.emp_id,e.e_name,e.sal,s.grade
from employee e, salgrade s where e.sal
between s.losal and s.hisal;

select e.e_name as Employee_name , m.e_name
as Manager_Name
from employee e left join employee m 
on e.mgr = m.emp_id;

select e.emp_id, e.e_name,d.dept_no,d.dept_name
from employee e cross join Dept d;

select e.emp_id , e.e_name , d.dept_no, d.loc
